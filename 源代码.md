## 4. logger.py - 日志模块

```python
  1→# -*- coding: utf-8 -*-
  2→"""
  3→校园体育器材出入库管理系统
  4→日志管理模块
  5→"""
  6→
  7→import os
  8→import logging
  9→from logging.handlers import RotatingFileHandler
 10→from datetime import datetime
 11→
 12→import config
 13→
 14→class Logger:
 15→    _loggers = {}
 16→    _initialized = False
 17→
 18→    @classmethod
 19→    def get_logger(cls, name='SportsEquipment'):
 20→        if name in cls._loggers:
 21→            return cls._loggers[name]
 22→
 23→        if not cls._initialized:
 24→            cls._setup_log_dir()
 25→            cls._initialized = True
 26→
 27→        logger = logging.getLogger(name)
 28→        logger.setLevel(getattr(logging, config.LOG_CONFIG['log_level']))
 29→
 30→        if not logger.handlers:
 31→            log_file = os.path.join(config.LOG_CONFIG['log_dir'], config.LOG_CONFIG['log_file'])
 32→            handler = RotatingFileHandler(
 33→                log_file,
 34→                maxBytes=config.LOG_CONFIG['max_bytes'],
 35→                backupCount=config.LOG_CONFIG['backup_count'],
 36→                encoding='utf-8'
 37→            )
 38→            formatter = logging.Formatter(
 39→                '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
 40→            )
 41→            handler.setFormatter(formatter)
 42→            logger.addHandler(handler)
 43→
 44→        cls._loggers[name] = logger
 45→        return logger
 46→
 47→    @classmethod
 48→    def _setup_log_dir(cls):
 49→        log_dir = os.path.join(config.BASE_DIR, config.LOG_CONFIG['log_dir'])
 50→        if not os.path.exists(log_dir):
 51→            os.makedirs(log_dir)
 52→
```
